--- ВОПРОС 1: СБОРКА ОБРАЗА И ЗАПУСК КОНТЕЙНЕРА ---

# Сборка образа из Dockerfile (выполнять в корне проекта)
docker build -t akcelerator/qt_tcp_server:latest .

# На Mac с Apple Silicon, если нужна linux/amd64:
docker build --platform linux/amd64 -t akcelerator/qt_tcp_server:latest .

# Запуск контейнера: порт 5555, сервер стартует сам
docker run --rm -it -p 5555:5555 akcelerator/qt_tcp_server:latest

# Публикация на Docker Hub (публичный доступ)
docker login
docker push akcelerator/qt_tcp_server:latest

--- ВОПРОС 2: ЛОКАЛЬНАЯ СБОРКА СЕРВЕРА И КЛИЕНТА ---

cd server
qmake qt_tcp_server.pro
make
./qt_tcp_server

cd client
qmake qt_tcp_client.pro
make
./qt_tcp_client

# Тест без GUI (несколько терминалов):
nc 127.0.0.1 5555
/add молоко
/add хлеб
/remove молоко
/finish
